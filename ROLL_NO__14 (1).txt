import numpy as np  
import matplotlib.pyplot as plt  


# Define a simple linear neural network model (y = Wx + b)  
class SimpleLinearNN:  


def __init__(self, input_size, output_size, learning_rate=0.01):  
# Initialize weights (W) and bias (b) randomly  
self.W = np.random.randn(input_size, output_size)  
self.b = np.zeros((1, output_size))  
self.lr = learning_rate  
self.loss_history = []  


def forward(self, X):  
"""Compute output y = XW + b"""  
return np.dot(X, self.W) + self.b  


def compute_loss(self, y_pred, y_true):  
"""Mean Squared Error"""  
return np.mean((y_pred - y_true) ** 2)  


def backward(self, X, y_true, y_pred):  
"""Update weights using gradient descent"""  
m = X.shape[0] # number of samples  


dW = (2 / m) * np.dot(X.T, (y_pred - y_true))  
db = (2 / m) * np.sum(y_pred - y_true, axis=0, keepdims=True)  


# Update parameters  
self.W -= self.lr * dW  
self.b -= self.lr * db  


def train(self, X, y, epochs=100):  
"""Training loop"""  
for epoch in range(epochs):  


# Forward pass  
y_pred = self.forward(X)  


# Compute loss  
loss = self.compute_loss(y_pred, y)  
self.loss_history.append(loss)  


# Backward pass  
self.backward(X, y, y_pred)  


# Print progress  
if (epoch + 1) % 20 == 0:  


print(f"Epoch {epoch+1}/{epochs}, Loss: {loss:.4f}")  


def plot_loss(self):  
"""Plot training loss over epochs"""  
plt.plot(self.loss_history, label="Training Loss")  
plt.xlabel("Epochs")  
plt.ylabel("Loss (MSE)")

import numpy as np  


# Sigmoid activation function and its derivative  
def sigmoid(x):  
    return 1 / (1 + np.exp(-x))  


def sigmoid_derivative(x):  
    return x * (1 - x)  


# XOR input and expected output  
X = np.array([[0,0], [0,1], [1,0], [1,1]])  
y = np.array([[0], [1], [1], [0]])  


# Seed for reproducibility  
np.random.seed(42)  


# Initialize weights and biases  
input_layer_neurons = 2  
hidden_layer_neurons = 2  
output_neurons = 1  


# Weights  
wh = np.random.uniform(size=(input_layer_neurons, hidden_layer_neurons))  
bh = np.random.uniform(size=(1, hidden_layer_neurons))  
wout = np.random.uniform(size=(hidden_layer_neurons, output_neurons))  
bout = np.random.uniform(size=(1, output_neurons))  


learning_rate = 0.5  
epochs = 10000  


# Training loop  
# Training loop  
for epoch in range(epochs):  
    # Forward Prop  
    hidden_layer_input = np.dot(X, wh) + bh  
    hidden_layer_activations = sigmoid(hidden_layer_input)  
      
    output_layer_input = np.dot(hidden_layer_activations, wout) + bout  
    output = sigmoid(output_layer_input)  






 
      
    # Backpropagation  
    error = y - output  
    d_output = error * sigmoid_derivative(output)  
      
    error_hidden_layer = d_output.dot(wout.T)  
    d_hidden_layer = error_hidden_layer * sigmoid_derivative(hidden_layer_activations)  
      
    # Updating Weights and Biases  
    wout += hidden_layer_activations.T.dot(d_output) * learning_rate  
    bout += np.sum(d_output, axis=0, keepdims=True) * learning_rate  
    wh += X.T.dot(d_hidden_layer) * learning_rate  
    bh += np.sum(d_hidden_layer, axis=0, keepdims=True) * learning_rate  


print("Final Predictions:")  
print(output)

import numpy as np  
import matplotlib.pyplot as plt  


class RBFNetwork:  
    def __init__(self, num_centers, sigma=1.0):  
        self.num_centers = num_centers  
        self.sigma = sigma  
        self.centers = None  
        self.weights = None  


    def _gaussian_rbf(self, x, center):  
        return np.exp(-np.linalg.norm(x - center)**2 / (2 * self.sigma**2))  


    def _calculate_interpolation_matrix(self, X):  
        G = np.zeros((X.shape[0], self.num_centers))  
        for i, x in enumerate(X):  
            for j, c in enumerate(self.centers):  
                G[i, j] = self._gaussian_rbf(x, c)  
        return G  


    def fit(self, X, y):  
        # Randomly choose centers from the training data  
        random_indices = np.random.choice(X.shape[0], self.num_centers, replace=False)  
        self.centers = X[random_indices]  
          
        # Calculate the interpolation matrix  
        G = self._calculate_interpolation_matrix(X)  
          
        # Solve for weights using the Moore-Penrose pseudo-inverse  
        self.weights = np.dot(np.linalg.pinv(G), y)

import numpy as np  


class HopfieldNetwork:  
    def __init__(self, size):  
        self.size = size  
        self.weights = np.zeros((size, size))  


    def train(self, patterns):  
        """  
        Train network on given patterns using Hebbian learning.  
        patterns: list or array of binary patterns (-1, 1)  
        """  
        for p in patterns:  
            p = p.reshape(self.size, 1)  
            self.weights += p @ p.T  


        np.fill_diagonal(self.weights, 0)  # no self-connection  


    def predict(self, pattern, max_iterations=10):  
        """  
        Recall a pattern from the network.  
        pattern: initial input pattern (-1, 1)  
        max_iterations: maximum updates before stopping  
        """  
        s = pattern.copy()  


        for _ in range(max_iterations):  
            for i in range(self.size):  
                raw = np.dot(self.weights[i], s)  
                s[i] = 1 if raw >= 0 else -1  


        return s

    ranked_pop = rank_routes(current_gen)  


    selection_results = selection(  
        ranked_pop,  
        elite_size  
    )  


    matingpool = mating_pool(  
        current_gen,  
        selection_results  
    )  


    children = breed_population(  
        matingpool,  
        elite_size  
    )  


    next_gen = mutate_population(  
        children,  
        mutation_rate  
    )  


    return next_gen
